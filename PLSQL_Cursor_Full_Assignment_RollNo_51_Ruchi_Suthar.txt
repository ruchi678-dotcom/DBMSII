-- File: 51_Section2Q1.sql
-- Display book_id, title and price of every book.
-- Use OPEN / FETCH / CLOSE and %NOTFOUND.

DECLARE
    CURSOR c_book IS
        SELECT book_id, title, price
        FROM book;

    v_book_id book.book_id%TYPE;
    v_title book.title%TYPE;
    v_price book.price%TYPE;

BEGIN
    OPEN c_book;

    LOOP
        FETCH c_book INTO v_book_id, v_title, v_price;

        EXIT WHEN c_book%NOTFOUND;

        DBMS_OUTPUT.PUT_LINE(
            'ID: ' || v_book_id ||
            ' | Title: ' || v_title ||
            ' | Price: ' || v_price
        );
    END LOOP;

    CLOSE c_book;
END;
/

-------------------- Q7 --------------------
-- File: 51_Section2Q7.sql
-- Check %ISOPEN before opening and after closing.

DECLARE
    CURSOR c_book IS
        SELECT book_id, title
        FROM book;

    v_book_id book.book_id%TYPE;
    v_title book.title%TYPE;

BEGIN
    IF c_book%ISOPEN THEN
        DBMS_OUTPUT.PUT_LINE('Cursor already open');
    ELSE
        DBMS_OUTPUT.PUT_LINE('Cursor is not open');
    END IF;

    OPEN c_book;

    LOOP
        FETCH c_book INTO v_book_id, v_title;

        EXIT WHEN c_book%NOTFOUND;
    END LOOP;

    CLOSE c_book;

    IF c_book%ISOPEN THEN
        DBMS_OUTPUT.PUT_LINE('Cursor is still open');
    ELSE
        DBMS_OUTPUT.PUT_LINE('Cursor is FALSE after CLOSE');
    END IF;
END;

-- File: 51_Section2Q8.sql
-- Display issues where return_date IS NULL.
-- If no rows: All books returned.

DECLARE
    CURSOR c_issue IS
        SELECT issue_id, book_id, issue_date
        FROM book_issue
        WHERE return_date IS NULL;

    v_issue_id book_issue.issue_id%TYPE;
    v_book_id book_issue.book_id%TYPE;
    v_issue_date book_issue.issue_date%TYPE;

BEGIN
    OPEN c_issue;

    FETCH c_issue INTO v_issue_id, v_book_id, v_issue_date;

    IF c_issue%NOTFOUND THEN
        DBMS_OUTPUT.PUT_LINE('All books returned');
    ELSE
        LOOP
            DBMS_OUTPUT.PUT_LINE(
                'Issue ID: ' || v_issue_id ||
                ' | Book ID: ' || v_book_id ||
                ' | Issue Date: ' ||
                TO_CHAR(v_issue_date, 'DD-MON-YYYY')
            );

            FETCH c_issue INTO v_issue_id, v_book_id, v_issue_date;

            EXIT WHEN c_issue%NOTFOUND;
        END LOOP;
    END IF;

    CLOSE c_issue;
END;

-- File: 51_Section2Q12.sql
-- Increase stock of Database books by 10.

DECLARE
    CURSOR c_book IS
        SELECT book_id, title, stock
        FROM book
        WHERE category = 'Database'
        FOR UPDATE;

    v_book_id book.book_id%TYPE;
    v_title book.title%TYPE;
    v_old_stock book.stock%TYPE;
    v_new_stock book.stock%TYPE;

BEGIN
    OPEN c_book;

    LOOP
        FETCH c_book INTO v_book_id, v_title, v_old_stock;

        EXIT WHEN c_book%NOTFOUND;

        v_new_stock := v_old_stock + 10;

        UPDATE book
        SET stock = v_new_stock
        WHERE CURRENT OF c_book;

        DBMS_OUTPUT.PUT_LINE(
            v_title ||
            ' | Old Stock: ' || v_old_stock ||
            ' | New Stock: ' || v_new_stock
        );
    END LOOP;

    CLOSE c_book;

    COMMIT;
END;

-- File: 51_Section3Q10.sql
-- Accept number of days.
-- Fine = Rs. 2 per overdue day.

DECLARE
    CURSOR c_issue(p_days NUMBER) IS
        SELECT issue_id,
               book_id,
               issue_date
        FROM book_issue
        WHERE return_date IS NULL
        AND TRUNC(SYSDATE) - issue_date > p_days;

    v_days NUMBER;
    v_fine NUMBER;
    v_total NUMBER := 0;

BEGIN
    FOR r IN c_issue(&days)
    LOOP
        v_days := TRUNC(SYSDATE) - r.issue_date;
        v_fine := v_days * 2;
        v_total := v_total + v_fine;

        DBMS_OUTPUT.PUT_LINE(
            'Issue ID: ' || r.issue_id ||
            ' | Book ID: ' || r.book_id ||
            ' | Overdue Days: ' || v_days ||
            ' | Fine: Rs. ' || v_fine
        );
    END LOOP;

    DBMS_OUTPUT.PUT_LINE(
        'Total Fine Payable: Rs. ' || v_total
    );
END;
/

-- Original issue:
-- c_book is a parameterised cursor used by a cursor FOR loop.
-- The FOR loop automatically opens and closes the cursor.
-- Therefore c_book%ROWCOUNT after the loop and CLOSE c_book
-- are invalid because the cursor has already been closed.
--

-- Correct version:

DECLARE
    CURSOR c_book (p_cat VARCHAR2) IS
        SELECT book_id, title
        FROM book
        WHERE category = p_cat;

    v_count NUMBER := 0;

BEGIN
    FOR r IN c_book('Database')
    LOOP
        v_count := v_count + 1;

        DBMS_OUTPUT.PUT_LINE(
            r.book_id || ' - ' || r.title
        );
    END LOOP;

    IF v_count = 0 THEN
        DBMS_OUTPUT.PUT_LINE('No books found');
    END IF;
END;

-------------------- Q4 --------------------
-- If a parameterised cursor has a required parameter and it is
-- opened without an argument, Oracle raises an error because the
-- required parameter was not supplied.
--
-- This is an execution-time error when the OPEN statement runs.
--
-- If the cursor parameter has a DEFAULT value, the default value
-- is used when the cursor is opened without an argument.
--
-- Example:
--
-- CURSOR c_book(

-------------------- Q5 --------------------
-- Comparison:
--
-- Implicit Cursor:
-- Who declares it: Oracle
-- Who opens/closes it: Oracle
-- Reused with different values: Not applicable
-- Best choice: Simple INSERT, UPDATE, DELETE or SELECT INTO.
--
-- Explicit Simple Cursor:
-- Who declares it: Programmer
-- Who opens/closes it: Programmer
-- Reused with different values: No parameters, so fixed query
-- Best choice: Processing multiple rows from one fixed query.
--
-- Parameterised Cursor:
-- Who declares it: Programmer
-- Who opens/closes it: Programmer or cursor FOR loop
-- Reused with different values: Yes
-- Best choice: Same query required for different input values.
